import json
from datetime import datetime, timedelta
import logging

class TaskQueue:
    def __init__(self, max_retries=3, retry_delay=1.0, persist_path=None):
        self.max_retries = max_retries
        self.retry_delay = retry_delay
        selfpersist_path = persist_path
        self.tasks = {}
        self.last_execution_time = None

        # Configuración de logging
        logging.basicConfig(level=logging.INFO)
        self.logger = logging.getLogger(__name__)

    def add_task(self, name: str, function: callable, args: tuple, kwargs: dict) -> str:
        """ agrega una nueva tarea al cuaderno """
        task_id = str(uuid.uuid4()) if not name else name
        start_time = datetime.now()
        
        try:
            self.tasks[task_id] = {
                'name': name,
                'function': function,
                'args': args,
                'kwargs': kwargs,
                'start_time': start_time,
                'status': 'pending',
                'logs': []
            }
            
            if self.persist_path is not None:
                with open(self.persist_path, 'w') as f:
                    json.dump(self.tasks, f)
                    
            return task_id
        except Exception as e:
            self.logger.error(f"Error al agregar tarea: {str(e)}")
            raise

    def run_all(self) -> None:
        """ ejecute todas las tareas con reintentos """
        try:
            for _ in range(self.max_retries):
                if not self.tasks:
                    break
                
                task_ids = list(self.tasks.keys())
                
                for task_id in task_ids.copy():
                    try:
                        start_time = datetime.now()
                        
                        result = None
                        status = 'pending'
                        logs = []
                        
                        try:
                            result = self._run_task(task_id)
                            status = 'completed'
                            
                        except Exception as e:
                            self.logger.error(f"Error ejecutando tarea {task_id}: {str(e)}")
                            status = 'failed'
                            
                        if status == 'completed':
                            logs.append({'timestamp': datetime.now(), 'result': result})
                        else:
                            logs.append({'timestamp': datetime.now(), 'error': str(e)})
                        
                        self.tasks[task_id]['status'] = status
                        self.tasks[task_id]['logs'] = logs
                        
                    except KeyboardInterrupt:
                        self.logger.info("Interceptado por interrupción del usuario")
                        break
                    
                else:
                    break  # Si no hubo interrupción, salta al siguiente bucle outer

        except Exception as e:
            self.logger.error(f"Error en run_all: {str(e)}")

    def _run_task(self, task_id) -> Any:
        """ Ejecuta una tarea específica """
        if task_id not in self.tasks or 'function' not in self.tasks[task_id]:
            raise ValueError(f"Tarea {task_id} no existe")
            
        try:
            start_time = datetime.now()
            result = self.tasks[task_id]['function'](*(self.tasks[task_id]['args'],))
            end_time = datetime.now()
            
            duration = (end_time - start_time).total_seconds()
            
            logs = [
                {
                    'timestamp': start_time,
                    'status': 'started',
                    'duration': duration
                },
                {
                    'timestamp': end_time,
                    'status': 'completed'
                }
            ]
            
            self.tasks[task_id]['logs'].append(logs)
            return result
            
        except Exception as e:
            error_log = {
                'timestamp': datetime.now(),
                'error': str(e),
                'task_id': task_id
            }
            self.tasks[task_id]['logs'].append(error_log)
            raise

    def run_one(self, task_id: str) -> dict:
        """ Ejecuta una sola tarea """
        try:
            if task_id not in self.tasks or 'function' not in self.tasks[task_id]:
                raise ValueError(f"Tarea {task_id} no existe")
                
            result = None
            logs = []
            
            start_time = datetime.now()
            try:
                result = self.tasks[task_id]['function'](*(self.tasks[task_id]['args'],))
                end_time = datetime.now()
                
                duration = (end_time - start_time).total_seconds()
                
                logs.append({
                    'timestamp': start_time,
                    'duration': duration
                })
                logs.append({'timestamp': end_time, 'status': 'completed'})
                
            except Exception as e:
                logs.append({
                    'timestamp': datetime.now(),
                    'error': str(e)
                })
                
            self.tasks[task_id]['logs'] = logs
            
            return {
                'task_id': task_id,
                'result': result,
                'status': 'completed'
            }
            
        except Exception as e:
            self.logger.error(f"Error ejecutando tarea {task_id}: {str(e)}")
            raise

    def get_status(self, task_id: str) -> dict:
        """ Obtiene el estado actual de una tarea """
        if task_id not in self.tasks or 'logs' not in self.tasks[task_id]:
            return None
            
        logs = self.tasks[task_id]['logs']
        
        last_log = next(reversed(logs), None)
        if last_log and last_log['status'] == 'completed':
            completed_at = last_log['timestamp']
            
            duration = (datetime.now() - completed_at).total_seconds()
            
            return {
                'task_id': task_id,
                'name': self.tasks[task_id]['name'],
                'status': 'completed',
                'duration': duration
            }
        
        failed_log = next((log for log in logs if log['error']), None)
        if failed_log:
            return {
                'task_id': task_id,
                'name': self.tasks[task_id]['name'],
                'status': 'failed',
                'error': failed_log['error']
            }
        
        pending_log = next((log for log in logs if log['status'] == 'started'), None)
        if pending_log:
            return {
                'task_id': task_id,
                'name': self.tasks[task_id]['name'],
                'status': 'pending',
                'duration': pending_log['duration']
            }
        
        return None

    def summary(self) -> dict:
        """ Resumen de las tareas """
        stats = {
            'total_tasks': len(self.tasks),
            'completed_tasks': 0,
            'pending_tasks': 0,
            'failed_tasks': 0,
            'total_time': 0.0
        }
        
        for task_id in self.tasks:
            task = self.tasks[task_id]
            logs = task['logs']
            
            if logs and logs[-1]['status'] == 'completed':
                stats['completed_tasks'] += 1
                
                completed_at = logs[-2]['timestamp']
                duration = (datetime.now() - completed_at).total_seconds()
                
                stats['total_time'] += duration
                
            elif logs:
                stats['pending_tasks'] += 1
                stats['failed_tasks'] += 1
        
        return stats

    def persist(self) -> None:
        """ Persiste los datos en el archivo JSON """
        if self.persist_path is not None and len(self.tasks) > 0:
            try:
                with open(selfpersist_path, 'w') as f:
                    json.dump(self.tasks, f)
            except Exception as e:
                self.logger.error(f"Error persistir: {str(e)}")
    
    @classmethod
    def load(cls, path: str) -> dict:
        """ Carga los datos desde el archivo JSON """
        try:
            with open(path, 'r') as f:
                tasks = json.load(f)
                
            queue = cls()
            for task_id in tasks:
                if 'logs' not in tasks[task_id] or len(tasks[task_id]['logs']) == 0:
                    continue
                
                logs = tasks[task_id]['logs']
                last_log = next(reversed(logs), None)
                if not last_log or last_log['status'] != 'completed':
                    continue
                    
                task = {
                    'name': None,
                    'function': None,
                    'args': tuple(),
                    'kwargs': dict()
                }
                
                for log in logs:
                    if log['status'] == 'started':
                        task['start_time'] = str(log['timestamp'])
                        task['duration'] = str((log['end_time'] - log['start_time']).total_seconds())
                    elif log['status'] == 'completed':
                        task['end_time'] = str(log['timestamp'])
                        task['result'] = str(log.get('result', ''))
                        task['error'] = str(log.get('error', ''))

                if name := next((k for k, v in tasks[task_id].items() if isinstance(v, dict) and 'name' in v), None):
                    task['name'] = name
                
                queue.tasks[task_id] = task
            
            return queue
        
        except Exception as e:
            self.logger.error(f"Error al cargar el cuaderno: {str(e)}")
            raise
